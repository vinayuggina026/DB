# EXP-7: Sqoop + Hive + HDFS Data Pipeline

## STEP 1: MySQL Setup

Login to MySQL:
mysql -u root -p

Create database and table:
CREATE DATABASE company;
USE company;

CREATE TABLE employee (
    emp_id INT PRIMARY KEY,
    name VARCHAR(50),
    dept VARCHAR(50),
    salary INT
);

INSERT INTO employee VALUES
(1,'Amit','DS',50000),
(2,'Ravi','HR',40000),
(3,'Sita','DS',60000),
(4,'John','Finance',45000),
(5,'Neha','HR',42000),
(6,'Kiran','DS',70000);

--------------------------------------------------

## STEP 2: Import DS Data into HDFS (Sqoop)

sqoop import \
--connect jdbc:mysql://localhost/company \
--username root \
--password root \
--query "SELECT emp_id,name,dept,salary FROM employee WHERE dept='DS' AND \$CONDITIONS" \
--target-dir /user/cloudera/emp_ds_data \
--split-by emp_id \
--m 1

--------------------------------------------------

## STEP 3: Verify HDFS Data

hdfs dfs -ls /user/cloudera/emp_ds_data

hdfs dfs -cat /user/cloudera/emp_ds_data/part-m-00000

Output:
1,Amit,DS,50000
3,Sita,DS,60000
6,Kiran,DS,70000

--------------------------------------------------

## STEP 4: Hive External Table

hive

CREATE EXTERNAL TABLE emp_ds (
    emp_id INT,
    name STRING,
    dept STRING,
    salary INT
)
ROW FORMAT DELIMITED
FIELDS TERMINATED BY ','
STORED AS TEXTFILE
LOCATION '/user/cloudera/emp_ds_data';

--------------------------------------------------

## STEP 5: Hive Analysis

CREATE TABLE emp_summary_ds AS
SELECT dept, AVG(salary) AS avg_salary
FROM emp_ds
GROUP BY dept;

SELECT * FROM emp_summary_ds;

Output:
DS 60000

exit;

--------------------------------------------------

## STEP 6: MySQL Target Table

USE company;

CREATE TABLE emp_summary_ds (
    dept VARCHAR(50),
    avg_salary FLOAT
);

--------------------------------------------------

## STEP 7: Export to MySQL

sqoop export \
--connect jdbc:mysql://localhost/company \
--username root \
--password root \
--table emp_summary_ds \
--export-dir /user/hive/warehouse/emp_summary_ds \
--input-fields-terminated-by '\001' \
--m 1

--------------------------------------------------

## STEP 8: Verify in MySQL

SELECT * FROM emp_summary_ds;

Output:
DS | 60000
